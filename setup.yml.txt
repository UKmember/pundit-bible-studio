name: 1. Set up (run once)
on: workflow_dispatch
jobs:
  setup:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 20 }
      - run: node jobs/setup.mjs
        env:
          SUPABASE_PROJECT_REF: ${{ vars.SUPABASE_PROJECT_REF }}
          SUPABASE_ACCESS_TOKEN: ${{ secrets.SUPABASE_ACCESS_TOKEN }}
          ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}
          OWNER_EMAIL: ${{ secrets.OWNER_EMAIL }}
          OWNER_PASSWORD: ${{ secrets.OWNER_PASSWORD }}
          SCAN_TOKEN: ${{ secrets.SCAN_TOKEN }}
          CLAUDE_MODEL: ${{ vars.CLAUDE_MODEL }}
          FOOTBALL_DATA_KEY: ${{ secrets.FOOTBALL_DATA_KEY }}
          HIGGSFIELD_KEY: ${{ secrets.HIGGSFIELD_KEY }}
          OPUSCLIP_KEY: ${{ secrets.OPUSCLIP_KEY }}
          HF_BUDGET_GBP: ${{ vars.HF_BUDGET_GBP }}
          MONTHLY_BUDGET_GBP: ${{ vars.MONTHLY_BUDGET_GBP }}
          ALL_SECRETS: ${{ toJSON(secrets) }}
          ALL_VARS: ${{ toJSON(vars) }}
